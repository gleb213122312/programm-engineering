```python
def tuple_sort(tpl):
    for elm in tpl:
        if not isinstance(elm, int):
            return tpl
    return tuple(sorted(tpl))

if __name__ == '__main__':
```

## Самостоятельная работа №2
### Функция, которая удаляет первое вхождение заданного элемента из кортежа и возвращает обновлённый кортеж.Если элемента нет — вернуть кортеж без изменений.Входные данные:
- (1, 2, 3), 1)
- (1, 2, 3, 1, 2, 3, 4, 5, 2, 3, 4, 2, 4, 2), 3)
- (2, 4, 6, 6, 4, 2), 9)
### Ожидаемый результат:
- (2, 3)
- (1, 2, 1, 2, 3, 4, 5, 2, 3, 4, 2, 4, 2)
- (2, 4, 6, 6, 4, 2)

```python
def remove_first_occurrence(t, value):
    try:
        i = t.index(value)
        return t[:i] + t[i+1:]
    except ValueError:
        return t

print(remove_first_occurrence((1, 2, 3), 1))
print(remove_first_occurrence((1, 2, 3, 1, 2, 3, 4, 5, 2, 3, 4, 2, 4, 2), 3))
print(remove_first_occurrence((2, 4, 6, 6, 4, 2), 9))
```

## Вывод:
Программа удаляет первое вхождение заданного элемента из кортежа и возвращает новый кортеж.
Для этого используется метод index() для поиска элемента, оператор среза (t[:i] + t[i+1:])
и конструкция try-except для обработки ситуации, когда элемента нет в кортеже.
  
## Самостоятельная работа №3
### Функция, которая принимает строку из цифр и формирует словарь с количеством повторений каждой.Функция должна вернуть словарь из трёх самых частых цифр и вывести их количества в порядке возрастания ключей.

```python
def most_common_digits(digits_str):
    counts = {}

    for ch in digits_str:
        num = int(ch)
        if num in counts:
            counts[num] += 1
        else:
            counts[num] = 1

    sorted_counts = sorted(counts.items(), key=lambda x: x[1], reverse=True)
    top3 = dict(sorted_counts[:3])

    print("Строка:", digits_str)
    print("Три самых популярных цифры:", list(top3.keys()))

    print("Значения по возрастанию ключа:")
    for key in sorted(top3):
        print(f"{key}: {top3[key]}")

    print("Словарь:", top3)


digits = "012345678909876543210123456789000"

most_common_digits(digits)
```

## Вывод:
Программа анализирует строку цифр, подсчитывает, сколько раз встречается каждая цифра,
и с помощью словаря, цикла for, функции sorted() и метода dict()
определяет три самые часто встречающиеся цифры, выводя их в порядке возрастания ключей.

## Самостоятельная работа №4
### Функция, которая принимает кортеж и элемент (id) и возвращает:
- часть кортежа от первого до второго появления элемента включительно;
- если элемент встречается один раз — срез от него до конца;
- если элемента нет — пустой кортеж.
### Входные данные:
- (1, 2, 3), 8)
- (1, 8, 3, 4, 8, 8, 9, 2), 8)
- (1, 2, 8, 5, 1, 2, 9), 8)

```python
def slice_between_first_two(t, value):
    indices = [i for i, v in enumerate(t) if v == value]
    if not indices:
        return tuple()
    if len(indices) == 1:
        return t[indices[0]:]
    return t[indices[0]:indices[1]+1]

print(slice_between_first_two((1, 2, 3), 8))
print(slice_between_first_two((1, 8, 3, 4, 8, 8, 9, 2), 8))
print(slice_between_first_two((1, 2, 8, 5, 1, 2, 9), 8))
```
